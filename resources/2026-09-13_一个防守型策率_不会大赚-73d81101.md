# 一个防守型策率，不会大赚

# -*- coding: utf-8 -*-
# ============================================================
# ETF低胜率高盈亏比趋势跟踪策略 — 聚宽版本（修复版）
# 直接粘贴到 joinquant.com 策略编辑器运行
# ============================================================

import numpy as np
import pandas as pd
from jqdata import *


# ============================================================
# 一、参数配置
# ============================================================
SYMBOLS = ['510300.XSHG', '510500.XSHG', '159915.XSHE']  # 标的池
BB_PERIOD = 20              # 布林带周期
BB_DEVFACTOR = 2.0          # 标准差倍数
ATR_PERIOD = 14             # ATR周期
ATR_STOP_MULT = 1.5         # 初始止损ATR倍数
ATR_TRAIL_MULT = 2.0        # 移动止盈ATR回撤倍数
MA_TREND_PERIOD = 60        # 趋势过滤均线周期
RISK_PER_TRADE = 0.005      # 单笔风险占总资金比例（0.5%）
MAX_POSITION_PCT = 0.30     # 单只ETF最大市值占比
MAX_HOLDINGS = 2            # 最多同时持仓数
MAX_CONSEC_LOSSES = 20      # 连续亏损暂停阈值
PAUSE_RECOVERY_DAYS = 5     # 空仓恢复天数


# ============================================================
# 二、初始化
# ============================================================
def initialize(context):
    # 基准：沪深300
    set_benchmark('000300.XSHG')
    # 前复权价格
    set_option('use_real_price', True)
    # ETF佣金：万分之2.5，最低5元
    set_order_cost(OrderCost(
        open_tax=0, close_tax=0,
        open_commission=0.00025, close_commission=0.00025,
        close_today_commission=0, min_commission=5
    ), type='fund')
    # 滑点0.1%
    set_slippage(FixedSlippage(0.001))
    # 日志：只显示error级订单信息
    log.set_level('order', 'error')

    # ===== 全局状态（g对象在模拟盘中持久化） =====
    g.symbols = SYMBOLS
    g.state = {}                # 各持仓标的的状态字典 {symbol: {...}}
    g.consecutive_losses = 0    # 连续亏损计数
    g.pause_days_left = 0       # 暂停开仓剩余天数
    g.trade_records = []        # 已完成的交易记录

    # 每日开盘执行主逻辑
    run_daily(trade_logic, time='open')


# ============================================================
# 三、指标计算工具函数
# ============================================================
def calc_indicators(symbol):
    """
    计算布林带、ATR、MA60
    注意：在开盘时刻调用，attribute_history返回截至前一交易日的数据
    close[-1]即为「昨日收盘」，close[-2]为「前日收盘」
    """
    lookback = max(MA_TREND_PERIOD, BB_PERIOD) + ATR_PERIOD + 5
    hist = attribute_history(
        symbol, lookback, '1d',
        ['open', 'high', 'low', 'close'],
        skip_paused=True, df=False
    )
    if len(hist['close']) &lt;  MA_TREND_PERIOD + 2:
        return None

    close = hist['close']
    high = hist['high']
    low = hist['low']

    # --- 布林带 ---
    ma20 = np.mean(close[-BB_PERIOD:])
    std20 = np.std(close[-BB_PERIOD:])
    bb_upper_yest = ma20 + BB_DEVFACTOR * std20          # 昨日上轨

    ma20_prev = np.mean(close[-BB_PERIOD - 1:-1])
    std20_prev = np.std(close[-BB_PERIOD - 1:-1])
    bb_upper_prev = ma20_prev + BB_DEVFACTOR * std20_prev  # 前日上轨

    # --- MA60 ---
    ma60 = np.mean(close[-MA_TREND_PERIOD:])

    # --- ATR ---
    trs = []
    for i in range(-ATR_PERIOD, 0):
        tr = max(
            high[i] - low[i],
            abs(high[i] - close[i - 1]),
            abs(low[i] - close[i - 1])
        )
        trs.append(tr)
    atr = np.mean(trs)

    return {
        'close_yest': close[-1],           # 昨日收盘
        'close_prev': close[-2],           # 前日收盘
        'bb_upper_yest': bb_upper_yest,    # 昨日布林带上轨
        'bb_upper_prev': bb_upper_prev,    # 前日布林带上轨
        'ma60': ma60,
        'atr': atr,
    }


# ============================================================
# 四、主交易逻辑（每个交易日开盘执行）
# ============================================================
def trade_logic(context):
    current_date = context.current_dt.date()
    current_data = get_current_data()

    # ==================== 1. 持仓管理（止损/止盈） ====================
    for symbol in list(context.portfolio.positions.keys()):
        pos = context.portfolio.positions[symbol]
        if pos.total_amount &lt; = 0:
            continue

        # 非本策略开仓的标的，忽略
        if symbol not in g.state:
            continue

        st = g.state[symbol]

        # 当前价：优先用最新价，退化到开盘价
        cur_price = current_data[symbol].last_price
        if cur_price is None or cur_price &lt; = 0 or np.isnan(cur_price):
            cur_price = current_data[symbol].day_open
        if cur_price is None or cur_price &lt; = 0:
            continue

        # 更新持仓期间最高价
        st['highest'] = max(st['highest'], cur_price)

        # 止损 / 移动止盈判断
        stop_hit = cur_price &lt; = st['stop_price']
        trail_hit = (st['highest'] - cur_price) &gt;= (ATR_TRAIL_MULT * st['atr'])

        if stop_hit or trail_hit:
            reason = '止损' if stop_hit else '移动止盈'
            # 市价单：下一tick成交
            order_target(symbol, 0)

            pnl = (cur_price - st['entry_price']) * st['size']
            g.trade_records.append({
                'symbol': symbol,
                'entry_date': st['entry_date'],
                'exit_date': current_date,
                'entry_price': st['entry_price'],
                'exit_price': cur_price,
                'size': st['size'],
                'pnl': pnl,
                'reason': reason,
            })

            # 更新连续亏损
            if pnl &lt;  0:
                g.consecutive_losses += 1
            else:
                g.consecutive_losses = 0

            log.info('[%s] %s 卖出@%.3f 数量=%d 盈亏=%.2f 连续亏损=%d' % (
                current_date, symbol, cur_price, st['size'], pnl,
                g.consecutive_losses
            ))
            del g.state[symbol]

    # ==================== 2. 暂停开仓检查 ====================
    if g.pause_days_left &gt; 0:
        g.pause_days_left -= 1
        if g.pause_days_left &lt; = 0:
            g.consecutive_losses = 0
            log.info('[%s] 空仓恢复期结束，恢复开仓' % current_date)
        return

    if g.consecutive_losses &gt;= MAX_CONSEC_LOSSES:
        # 达到阈值：仅当空仓时才启动恢复计时
        holdings = sum(1 for s in context.portfolio.positions
                       if context.portfolio.positions[s].total_amount &gt; 0)
        if holdings == 0:
            g.pause_days_left = PAUSE_RECOVERY_DAYS
            log.info('[%s] 连续亏损%d次，暂停开仓%d天' % (
                current_date, g.consecutive_losses, PAUSE_RECOVERY_DAYS))
        return

    # ==================== 3. 开仓信号 ====================
    current_holdings = sum(1 for s in context.portfolio.positions
                           if context.portfolio.positions[s].total_amount &gt; 0)
    if current_holdings &gt;= MAX_HOLDINGS:
        return

    for symbol in g.symbols:
        # 已持仓则跳过
        if symbol in context.portfolio.positions and \
                context.portfolio.positions[symbol].total_amount &gt; 0:
            continue

        ind = calc_indicators(symbol)
        if ind is None:
            continue

        # 四条件：昨日收盘突破布林上轨 + 上轨向上 + 站上MA60
        cond1 = ind['close_yest'] &gt; ind['bb_upper_yest']
        cond2 = ind['close_prev'] &lt; = ind['bb_upper_prev']
        cond3 = ind['bb_upper_yest'] &gt; ind['bb_upper_prev']
        cond4 = ind['close_yest'] &gt; ind['ma60']

        if not (cond1 and cond2 and cond3 and cond4):
            continue

        # 入场价：今日开盘价
        entry_price = current_data[symbol].day_open
        if entry_price is None or entry_price &lt; = 0:
            entry_price = ind['close_yest']

        stop_price = entry_price - ATR_STOP_MULT * ind['atr']
        risk_per_share = entry_price - stop_price
        if risk_per_share &lt; = 0:
            continue

        # --- 仓位计算 ---
        total_value = context.portfolio.total_value
        risk_amount = total_value * RISK_PER_TRADE
        size = int(risk_amount / risk_per_share)
        size = (size // 100) * 100

        # 资金上限：单只不超过30%
        max_size = int(total_value * MAX_POSITION_PCT / entry_price)
        max_size = (max_size // 100) * 100
        size = min(size, max_size)

        # 最小交易单位：不足100份则尝试买100份，否则放弃
        if size &lt;  100:
            if total_value * MAX_POSITION_PCT &gt;= entry_price * 100:
                size = 100
            else:
                continue

        # --- 下单 ---
        order(symbol, size)

        # --- 记录状态 ---
        g.state[symbol] = {
            'entry_price': entry_price,
            'stop_price': stop_price,
            'highest': entry_price,
            'atr': ind['atr'],
            'entry_date': current_date,
            'size': size,
        }

        log.info('[%s] 开仓 %s 数量=%d 入场价~%.3f 止损价~%.3f ATR=%.3f' % (
            current_date, symbol, size, entry_price, stop_price, ind['atr']
        ))

        current_holdings += 1
        if current_holdings &gt;= MAX_HOLDINGS:
            break


# ============================================================
# 五、回测结束统计
# ============================================================
def on_strategy_end(context):
    """回测结束时输出统计报告"""
    if not g.trade_records:
        log.info('============ 无交易记录 ============')
        return

    df = pd.DataFrame(g.trade_records)
    total_trades = len(df)
    wins = df[df['pnl'] &gt; 0]
    losses = df[df['pnl'] &lt; = 0]

    win_rate = len(wins) / total_trades * 100 if total_trades &gt; 0 else 0
    avg_win = wins['pnl'].mean() if len(wins) &gt; 0 else 0
    avg_loss = abs(losses['pnl'].mean()) if len(losses) &gt; 0 else 1
    profit_factor = avg_win / avg_loss if avg_loss &gt; 0 else 0

    # 最大连续亏损（按时间顺序）
    max_consec = 0
    cur_consec = 0
    for pnl in df['pnl'].values:
        if pnl &lt;  0:
            cur_consec += 1
            max_consec = max(max_consec, cur_consec)
        else:
            cur_consec = 0

    # 平均持仓天数（用日期差近似，交易日约0.7）
    try:
        df['entry_dt'] = pd.to_datetime(df['entry_date'])
        df['exit_dt'] = pd.to_datetime(df['exit_date'])
        avg_days = (df['exit_dt'] - df['entry_dt']).dt.days.mean()
    except Exception:
        avg_days = 0

    total_return = (context.portfolio.total_value /
                    context.portfolio.starting_cash - 1) * 100

    log.info('=' * 60)
    log.info('         ETF低胜率高盈亏比趋势跟踪策略 — 回测报告')
    log.info('=' * 60)
    log.info('  期末权益:       %.2f' % context.portfolio.total_value)
    log.info('  总收益率:       %.2f%%' % total_return)
    log.info('  交易次数:       %d' % total_trades)
    log.info('  盈利次数:       %d' % len(wins))
    log.info('  亏损次数:       %d' % len(losses))
    log.info('  胜率:           %.2f%%' % win_rate)
    log.info('  盈亏比:         %.2f' % profit_factor)
    log.info('  最大连续亏损:   %d 次' % max_consec)
    log.info('  平均持仓天数:   %.1f 天' % avg_days)
    log.info('=' * 60)

    # 交易明细
    log.info('--- 交易明细 ---')
    for _, row in df.iterrows():
        log.info('%s  %s→%s  入场%.3f 出场%.3f  盈亏%+.2f  (%s)' % (
            row['symbol'], row['entry_date'], row['exit_date'],
            row['entry_price'], row['exit_price'], row['pnl'], row['reason']
        ))
